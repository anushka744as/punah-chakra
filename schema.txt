================================================================================
                    PUNAHCHAKRA SUPABASE DATABASE SCHEMA
================================================================================

TABLE 1: public.profiles
--------------------------------------------------------------------------------
Fields:
  - id              : UUID PRIMARY KEY (REFERENCES auth.users.id ON DELETE CASCADE)
  - email           : TEXT NOT NULL
  - full_name       : TEXT NOT NULL
  - phone           : TEXT DEFAULT ''
  - address         : TEXT DEFAULT ''
  - role            : TEXT NOT NULL CHECK (role IN ('donor', 'collector', 'recycler'))
  - points          : INTEGER DEFAULT 0
  - co2_offset_kg   : NUMERIC(10, 2) DEFAULT 0.0
  - created_at      : TIMESTAMPTZ DEFAULT NOW()
  - updated_at      : TIMESTAMPTZ DEFAULT NOW()

RLS Policies:
  - Public read access for authenticated users.
  - Users can update their own profile rows.

--------------------------------------------------------------------------------
TABLE 2: public.donations
--------------------------------------------------------------------------------
Fields:
  - id              : UUID PRIMARY KEY DEFAULT gen_random_uuid()
  - donor_id        : UUID NOT NULL (REFERENCES public.profiles.id ON DELETE CASCADE)
  - category        : TEXT NOT NULL
  - weight_kg       : NUMERIC(10, 2) DEFAULT 0.0
  - pickup_day      : TEXT NOT NULL
  - time_slot       : TEXT NOT NULL
  - instructions    : TEXT DEFAULT ''
  - address         : TEXT NOT NULL
  - status          : TEXT DEFAULT 'scheduled' CHECK (status IN ('scheduled', 'assigned', 'collected', 'assessed', 'completed', 'cancelled'))
  - points_awarded  : INTEGER DEFAULT 0
  - created_at      : TIMESTAMPTZ DEFAULT NOW()
  - updated_at      : TIMESTAMPTZ DEFAULT NOW()

RLS Policies:
  - Donors can view and insert their own donations.
  - Collectors can view all scheduled/assigned donations.

--------------------------------------------------------------------------------
TABLE 3: public.pickups
--------------------------------------------------------------------------------
Fields:
  - id              : UUID PRIMARY KEY DEFAULT gen_random_uuid()
  - donation_id     : UUID (REFERENCES public.donations.id ON DELETE CASCADE)
  - collector_id    : UUID (REFERENCES public.profiles.id ON DELETE SET NULL)
  - donor_name      : TEXT NOT NULL
  - donor_phone     : TEXT DEFAULT ''
  - address         : TEXT NOT NULL
  - zone            : TEXT DEFAULT 'Noida Sector 15A & 44'
  - latitude        : NUMERIC(10, 6) DEFAULT 28.5833
  - longitude       : NUMERIC(10, 6) DEFAULT 77.3167
  - weight_summary  : TEXT DEFAULT '12.4 kg'
  - time_window     : TEXT DEFAULT '09:00 AM - 12:00 PM'
  - status          : TEXT DEFAULT 'pending' CHECK (status IN ('pending', 'in_transit', 'collected', 'assessed', 'completed'))
  - scheduled_date  : TEXT DEFAULT 'Today'
  - created_at      : TIMESTAMPTZ DEFAULT NOW()
  - updated_at      : TIMESTAMPTZ DEFAULT NOW()

RLS Policies:
  - Collectors can read and update pickup jobs.

--------------------------------------------------------------------------------
TABLE 4: public.category_points
--------------------------------------------------------------------------------
Fields:
  - id                : UUID PRIMARY KEY DEFAULT gen_random_uuid()
  - category          : TEXT UNIQUE NOT NULL
  - base_points       : INTEGER NOT NULL DEFAULT 100
  - points_per_kg     : INTEGER NOT NULL DEFAULT 50
  - co2_factor_per_kg : NUMERIC(6, 2) NOT NULL DEFAULT 1.5
  - description       : TEXT DEFAULT ''
  - created_at        : TIMESTAMPTZ DEFAULT NOW()
  - updated_at        : TIMESTAMPTZ DEFAULT NOW()

RLS Policies:
  - Public read access for all users (`ALLOW SELECT for all`).
  - Authenticated users can insert/update category point rules.

--------------------------------------------------------------------------------
TABLE 5: public.notifications
--------------------------------------------------------------------------------
Fields:
  - id              : UUID PRIMARY KEY DEFAULT gen_random_uuid()
  - user_id         : UUID NOT NULL (REFERENCES public.profiles.id ON DELETE CASCADE)
  - title           : TEXT NOT NULL
  - body            : TEXT NOT NULL
  - tag             : TEXT DEFAULT 'UPDATE'
  - is_read         : BOOLEAN DEFAULT FALSE
  - created_at      : TIMESTAMPTZ DEFAULT NOW()

RLS Policies:
  - Users can read and update their own notifications.
